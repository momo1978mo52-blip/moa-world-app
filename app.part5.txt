  }
  if(state.memberTab==='gallery'){
    const faceActual=images.filter(i=>i.category==='顔8方向').length,bodyActual=images.filter(i=>i.category==='体型8方向').length;
    const face=FACE_ANGLES.map(a=>slotHtml(a,images.find(i=>i.category==='顔8方向'&&i.angle===a))).join('');
    const body=BODY_ANGLES.map(a=>slotHtml(a,images.find(i=>i.category==='体型8方向'&&i.angle===a))).join('');

    return `<div class="section-title"><h2>顔8方向</h2><small>画像登録 ${faceActual}/8 ・進行記録 ${fr}/8</small></div><div class="angle-grid">${face}</div><div class="section-title"><h2>体型8方向</h2><small>画像登録 ${bodyActual}/8 ・進行記録 ${br}/8</small></div><div class="angle-grid">${body}</div><div class="section-title"><h2>アルバム</h2></div><div class="album-grid">${albums}</div>`;
  }
  if(state.memberTab==='relations') return relations.length?relations.map(r=>`<div class="card"><b>${esc(r.personA===m.id?r.personBName:r.personAName)}</b><div class="badges">${(r.tags||[]).map(t=>`<span class="badge">${esc(t)}</span>`).join('')}</div><div class="tiny muted" style="margin-top:8px">${[r.aToB,r.bToA].filter(Boolean).map(esc).join(' / ')}${r.closeness?` ・ 親密度 ${r.closeness}/5`:''}</div>${r.memo?`<div class="tiny muted" style="margin-top:6px">${esc(r.memo)}</div>`:''}</div>`).join(''):`<div class="empty">関係データはまだありません</div>`;

    const fa=images.filter(i=>i.category==='顔8方向'&&i.status?.startsWith('採用')).length,ba=images.filter(i=>i.category==='体型8方向'&&i.status?.startsWith('採用')).length;
    const f=Math.max(fa,Number(m.faceProgress)||0),b=Math.max(ba,Number(m.bodyProgress)||0);
    return `<div class="card"><b>顔8方向 ${f}/8 <small class="muted">（画像本体 ${fa}/8）</small></b><div class="progress"><i style="width:${f/8*100}%"></i></div><div style="height:12px"></div><b>体型8方向 ${b}/8 <small class="muted">（画像本体 ${ba}/8）</small></b><div class="progress"><i style="width:${b/8*100}%"></i></div><div class="setting-row"><span>ポスター</span><span>${esc(m.posterStatus||'未確認')}</span></div></div><div class="card"><b>次に生成する前の確認</b><div class="setting-row"><span>次カテゴリ</span><span>${esc(prod?.nextCategory||'未設定')}</span></div><div class="setting-row"><span>次角度</span><span>${esc(prod?.nextAngle||'未設定')}</span></div><div class="setting-row"><span>再評価メモ</span><span class="muted">${esc(prod?.recheckMemo||'未設定')}</span></div>${prod?.cautions?`<div class="setting-row"><span>注意</span><span class="muted">${esc(prod.cautions)}</span></div>`:''}<button class="gold-btn" data-action="production-edit">次の制作を設定</button></div>`;
  }
  return _baseMemberTabHtml(m,images,relations,schedules,prod);
};

seed=async function(){
  const current=await getAll('members');
  if(localStorage.getItem('moaWorldSeedVersion')===WORLD_SEED_VERSION){if(!current.length){/* fall through */}else return;}
  const byName=new Map(current.map(m=>[m.name,m]));
  for(const src of WORLD_MEMBERS){
    const old=byName.get(src.name);
    const obj=old?{...old,...src,id:old.id,createdAt:old.createdAt||now(),updatedAt:now(),deletedAt:old.deletedAt||null}:{...src,id:uid('mem'),createdAt:now(),updatedAt:now(),deletedAt:null};
    await put('members',obj);byName.set(src.name,obj);
  }
  for(const rel of WORLD_RELATIONS){
    const a=byName.get(rel.a),b=byName.get(rel.b);if(!a||!b)continue;
    await put('relations',{id:`seedrel_${rel.k}`,seedKey:rel.k,personA:a.id,personB:b.id,personAName:a.name,personBName:b.name,tags:rel.tags||[],aToB:rel.aToB||'',bToA:rel.bToA||'',closeness:null,memo:rel.memo||'',createdAt:now(),updatedAt:now()});
  }
  const currentProd=await getAll('production');
  for(const p of WORLD_PRODUCTION){const m=byName.get(p.name);if(!m)continue;const old=currentProd.find(x=>x.memberId===m.id);await put('production',{...(old||{}),id:old?.id||`prod_${m.id}`,memberId:m.id,nextCategory:p.nextCategory||'',nextAngle:p.nextAngle||'',primaryImageId:old?.primaryImageId||'',supplementalImageIds:old?.supplementalImageIds||[],recheckMemo:p.recheckMemo||'',cautions:p.cautions||'',updatedAt:now()});}
  localStorage.setItem('moaWorldSeedVersion',WORLD_SEED_VERSION);
  await log('既存データ移行','app','world-seed',`${WORLD_MEMBERS.length}名・関係${WORLD_RELATIONS.length}件を反映`);
};
