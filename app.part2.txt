function memberTabHtml(m,images,relations,schedules,prod){
  if(state.memberTab==='profile') return `<div class="card"><div class="setting-row"><span>ステータス</span><b>${esc(m.status||'候補')}</b></div><div class="setting-row"><span>性格</span><span class="muted">${esc(m.personality||'未設定')}</span></div><div class="setting-row"><span>趣味</span><span class="muted">${esc(m.hobbies||'未設定')}</span></div><div class="setting-row"><span>メモ</span><span class="muted">${esc(m.memo||'未設定')}</span></div></div>`;
  if(state.memberTab==='gallery'){
    const face=FACE_ANGLES.map(a=>slotHtml('顔8方向',a,images.find(i=>i.category==='顔8方向'&&i.angle===a))).join('');
    const body=BODY_ANGLES.map(a=>slotHtml('体型8方向',a,images.find(i=>i.category==='体型8方向'&&i.angle===a))).join('');
    const albums=ALBUMS.map(a=>{const xs=images.filter(i=>i.category===a);return `<div class="album"><b>${a}</b><span class="count">${xs.length}</span><small class="muted"> 枚</small></div>`}).join('');
    return `<div class="section-title"><h2>顔8方向</h2><small>${images.filter(i=>i.category==='顔8方向').length}/8</small></div><div class="angle-grid">${face}</div><div class="section-title"><h2>体型8方向</h2><small>${images.filter(i=>i.category==='体型8方向').length}/8</small></div><div class="angle-grid">${body}</div><div class="section-title"><h2>アルバム</h2></div><div class="album-grid">${albums}</div>`;
  }
  if(state.memberTab==='relations') return relations.length?relations.map(r=>`<div class="card"><b>${esc(r.personA===m.id?r.personBName:r.personAName)}</b><div class="badges">${(r.tags||[]).map(t=>`<span class="badge">${esc(t)}</span>`).join('')}</div><div class="tiny muted" style="margin-top:8px">${esc(r.aToB||'')} / ${esc(r.bToA||'')} ・ 親密度 ${r.closeness||3}/5</div></div>`).join(''):`<div class="empty">関係データはまだありません</div>`;
  if(state.memberTab==='schedule') return schedules.length?schedules.sort((a,b)=>(a.date+a.start).localeCompare(b.date+b.start)).map(s=>`<div class="card"><b>${esc(s.date)} ${esc(s.start)}–${esc(s.end)}</b><div class="muted tiny">${esc(s.content||'予定')} ${s.place?`・${esc(s.place)}`:''}</div></div>`).join(''):`<div class="empty">予定はまだありません</div>`;
  if(state.memberTab==='production'){
    const f=images.filter(i=>i.category==='顔8方向'&&i.status?.startsWith('採用')).length;const b=images.filter(i=>i.category==='体型8方向'&&i.status?.startsWith('採用')).length;
    return `<div class="card"><b>顔8方向 ${f}/8</b><div class="progress"><i style="width:${f/8*100}%"></i></div><div style="height:12px"></div><b>体型8方向 ${b}/8</b><div class="progress"><i style="width:${b/8*100}%"></i></div></div><div class="card"><b>次に生成する前の確認</b><div class="setting-row"><span>次カテゴリ</span><span>${esc(prod?.nextCategory||'未設定')}</span></div><div class="setting-row"><span>次角度</span><span>${esc(prod?.nextAngle||'未設定')}</span></div><div class="setting-row"><span>再評価メモ</span><span class="muted">${esc(prod?.recheckMemo||'未設定')}</span></div><button class="gold-btn" data-action="production-edit">次の制作を設定</button></div>`;
  }
  return '';
}

async function renderSchedule(){
  const members=await activeMembers();const all=await getAll('schedules');const d=state.month;const y=d.getFullYear(),mo=d.getMonth();const first=new Date(y,mo,1);const start=new Date(y,mo,1-first.getDay());const today=toDateKey(new Date());
  let days='';for(let i=0;i<42;i++){const x=new Date(start);x.setDate(start.getDate()+i);const key=toDateKey(x);const count=all.filter(e=>e.date===key&&e.state!=='cancelled').length;days+=`<button class="day ${x.getMonth()!==mo?'dim':''} ${key===today?'today':''} ${key===state.selectedDate?'selected':''}" data-date="${key}"><b>${x.getDate()}</b><div>${count?`<span class="dot"></span><small>${count}</small>`:''}</div></button>`}
  const selected=all.filter(e=>e.date===state.selectedDate).sort((a,b)=>a.start.localeCompare(b.start));
  app.innerHTML=`<div class="calendar-head"><button class="ghost-btn" id="prevMonth">‹</button><b>${y}年 ${mo+1}月</b><button class="ghost-btn" id="nextMonth">›</button></div><div class="calendar-grid">${['日','月','火','水','木','金','土'].map(x=>`<div class="dow">${x}</div>`).join('')}${days}</div><div class="section-title"><h2>${state.selectedDate}</h2><button class="gold-btn tiny" data-action="schedule-add">＋予定</button></div><div class="timeline">${selected.length?selected.map(e=>`<div class="event ${e.state||''}"><b>${esc(e.start)}–${esc(e.end)} ${esc(members.find(m=>m.id===e.memberId)?.name||'未指定')}</b><div class="tiny muted">${esc(e.content||'予定')} ${e.place?`・${esc(e.place)}`:''}</div></div>`).join(''):`<div class="empty">この日の予定はありません</div>`}</div>`;
  document.getElementById('prevMonth').onclick=()=>{state.month=new Date(y,mo-1,1);renderSchedule()};document.getElementById('nextMonth').onclick=()=>{state.month=new Date(y,mo+1,1);renderSchedule()};document.querySelectorAll('[data-date]').forEach(b=>b.onclick=()=>{state.selectedDate=b.dataset.date;renderSchedule()});bindCommon();
