// Relationship records are additive. Existing IDs, profiles, images and notes stay intact.
function bondNameKey(value){return String(value||'').normalize('NFKC').replace(/[\s　]+/g,'').toLowerCase();}
function bondMemberMatches(members,name){
  const spec=MOA_BOND_EDITION.members.find(m=>m.name===name);
  const keys=new Set([name,...(spec?.aliases||[])].map(bondNameKey));
  return members.filter(m=>keys.has(bondNameKey(m.name)));
}
function bondMarkerId(){return 'bondedition_'+MOA_BOND_EDITION.version;}

function prepareBondPlan(members,relations,historyRows,stamp){
  if(historyRows.some(h=>h.id===bondMarkerId()))return {done:true,memberWrites:[],relationWrites:[],issues:[]};
  const allMembers=members.slice(),memberWrites=[],relationWrites=[],issues=[];
  for(const src of MOA_BOND_EDITION.members){
    const matches=bondMemberMatches(allMembers,src.name);
    if(matches.length){if(matches.length>1)issues.push(src.name+'：同じ名前・別名の登録が複数あります');continue;}
    const {key,aliases,...profile}=src;
    const row={...profile,id:'bondmem_'+key,favorite:false,personality:'',hobbies:'',memo:'',createdAt:stamp,updatedAt:stamp,deletedAt:null,bondSeedKey:key};
    if(allMembers.some(m=>m.id===row.id)){issues.push(src.name+'：人物IDの確認が必要です');continue;}
    allMembers.push(row);memberWrites.push(row);
  }
  for(const src of MOA_BOND_EDITION.relations){
    const aMatches=bondMemberMatches(allMembers,src.a),bMatches=bondMemberMatches(allMembers,src.b);
    if(aMatches.length!==1||bMatches.length!==1){issues.push(src.a+'・'+src.b+'：人物を一意に確認できません');continue;}
    const a=aMatches[0],b=bMatches[0];
    if(a.deletedAt||b.deletedAt)continue;
    const pair=relations.filter(r=>(r.personA===a.id&&r.personB===b.id)||(r.personA===b.id&&r.personB===a.id));
    const old=pair.find(r=>r.seedKey===src.key)||pair.find(r=>!r.deletedAt)||pair[0];
    if(old?.deletedAt)continue;
    if(old?.bond?.editedAt)continue;
    const {key,a:sourceA,b:sourceB,tags,...record}=src;
    const base=old||{id:'bondrel_'+key,personA:a.id,personB:b.id,personAName:a.name,personBName:b.name,aToB:'',bToA:'',memo:'',closeness:null,createdAt:stamp};
    if(!old&&relations.some(r=>r.id===base.id)){issues.push(src.a+'・'+src.b+'：関係IDの確認が必要です');continue;}
    const previousTags=Array.isArray(base.tags)?base.tags:String(base.tags||'').split(',').filter(Boolean);
    relationWrites.push({...base,tags:[...new Set([...previousTags,...tags])],bond:{...record,key,edition:MOA_BOND_EDITION.version,recordedAt:stamp},updatedAt:stamp});
  }
  return {done:false,memberWrites,relationWrites,issues};
}

function migrateBondEdition(){
  return new Promise((resolve,reject)=>{
    const tx=db.transaction(['members','relations','history'],'readwrite');
    const rows={},names=['members','relations','history'];let remaining=names.length,result;
    tx.oncomplete=()=>resolve(result);
    tx.onabort=()=>reject(tx.error||new Error('関係情報の保存が完了しませんでした'));
    tx.onerror=()=>{};
    names.forEach(name=>{
      const req=tx.objectStore(name).getAll();
      req.onsuccess=()=>{
        rows[name]=req.result;
        if(--remaining)return;
        try{
          const stamp=now(),plan=prepareBondPlan(rows.members,rows.relations,rows.history,stamp);result=plan;
          if(plan.done)return;
          const historyStore=tx.objectStore('history');
          historyStore.put({id:'bondsnapshot_'+MOA_BOND_EDITION.version,action:'人物関係更新前の記録',targetType:'app',targetId:'world-bonds',detail:'人物と関係の更新前記録',changedAt:stamp,snapshot:{members:rows.members,relations:rows.relations}});
          plan.memberWrites.forEach(row=>tx.objectStore('members').put(row));
          plan.relationWrites.forEach(row=>tx.objectStore('relations').put(row));
          historyStore.put({id:bondMarkerId(),action:'人物関係と絆の反映',targetType:'app',targetId:'world-bonds',detail:plan.relationWrites.length+'件の関係と8つの出来事を反映',changedAt:stamp,issues:plan.issues});
        }catch(error){result=undefined;tx.abort();reject(error);}
      };
    });
  });
}

  catch(error){console.error('Relationship update failed',error);state.bondMigrationError=true;}

function bondTags(r){return Array.isArray(r.tags)?r.tags:String(r.tags||'').split(',').filter(Boolean);}
function bondPeople(r,members=[]){
  return [members.find(m=>m.id===r.personA)?.name||r.personAName||'人物A',members.find(m=>m.id===r.personB)?.name||r.personBName||'人物B'];
}
function bondLines(text){return esc(text||'').replace(/\n/g,'<br>');}
function bondRecordHtml(r,person=null,members=[]){
  const names=bondPeople(r,members),b=r.bond||{};
  const title=person?(person.id===r.personA?names[1]:names[0]):names.join(' × ');
  const part=(label,value)=>value?'<div><dt>'+esc(label)+'</dt><dd>'+bondLines(value)+'</dd></div>':'';
  const events=Array.isArray(b.events)?b.events:[];
  return '<article class="card bond-card">'+
    '<div class="bond-card-heading"><h3>'+esc(title)+'</h3><button type="button" class="ghost-btn tiny" data-bond-edit="'+esc(r.id)+'">編集</button></div>'+
    (b.label?'<div class="gold bond-label">'+esc(b.label)+'</div>':'')+
    '<div class="badges">'+bondTags(r).map(t=>'<span class="badge">'+esc(t)+'</span>').join('')+(b.status?'<span class="badge '+(b.status==='確認済み'?'ok':'warn')+'">'+esc(b.status)+'</span>':'')+'</div>'+
    '<dl class="bond-facts">'+part('関係の始まり',b.origin||'今回、詳しい始まりは未回収。')+
    '<div><dt>絆が深まった出来事</dt><dd>'+(events.length?'<ul>'+events.map(t=>'<li>'+bondLines(t)+'</li>').join('')+'</ul>':'具体的な出来事は未回収。')+'</dd></div>'+
    part('現在の距離感',b.distance||[r.aToB,r.bToA].filter(Boolean).join(' / ')||'未回収')+
    part('これからの構想',b.plans)+part('確認したいこと',b.unresolved)+'</dl>'+
    (r.memo?'<details class="bond-note"><summary>これまでのメモ</summary><p>'+bondLines(r.memo)+'</p></details>':'')+
    ([r.aToB,r.bToA].some(Boolean)?'<div class="tiny muted">'+[r.aToB,r.bToA].filter(Boolean).map(esc).join(' / ')+'</div>':'')+
    (r.closeness?'<div class="tiny muted">手入力の親密度 '+esc(r.closeness)+'/5</div>':'')+
    (b.source?'<details class="bond-note"><summary>記録の根拠'+(b.editedAt?'・編集済み':'')+'</summary><p>'+esc(b.source)+'</p></details>':'')+
    (!person?'<div class="actions">'+names.map((name,i)=>'<button type="button" class="ghost-btn tiny" data-bond-profile="'+esc(i?r.personB:r.personA)+'">'+esc(name)+'の関係</button>').join('')+'</div>':'')+'</article>';
}

function bondQueryMatches(text,query){return !query||bondNameKey(text).includes(bondNameKey(query));}
function bondVisibleRows(relations,members,personId,query){
  const ids=new Set(members.map(m=>m.id));
  return relations.filter(r=>!r.deletedAt&&ids.has(r.personA)&&ids.has(r.personB))
    .filter(r=>!personId||r.personA===personId||r.personB===personId)
    .filter(r=>bondQueryMatches(JSON.stringify([bondPeople(r,members),bondTags(r),r.bond||{},r.memo||'']),query))
    .sort((a,b)=>bondPeople(a,members).join('').localeCompare(bondPeople(b,members).join(''),'ja'));
}
function bondNetworkHtml(center,rows,members){
  if(!center)return '<div class="empty">人物がまだ登録されていません。</div>';
  const related=rows.filter(r=>r.personA===center.id||r.personB===center.id);
  if(!related.length)return '<div class="empty">'+esc(center.name)+'の関係は、この条件では見つかりません。</div>';
  return '<div class="bond-network" aria-label="'+esc(center.name)+'を中心とする人物相関">'+
    '<div class="bond-network-root"><button type="button" class="gold-btn" data-bond-profile="'+esc(center.id)+'">'+esc(center.name)+'</button></div>'+
    '<ul>'+related.map(r=>{
      const otherId=r.personA===center.id?r.personB:r.personA,other=members.find(m=>m.id===otherId);
      const label=r.bond?.label||bondTags(r).join('・')||'関係あり';
      return '<li><div class="bond-network-edge"><span>'+esc(label)+'</span></div><button type="button" class="ghost-btn" data-bond-person="'+esc(otherId)+'">'+esc(other?.name||'人物')+'</button></li>';
    }).join('')+'</ul></div><p class="tiny muted">人物名を押すと、その人を中心に切り替わります。</p>';
}
function bondStoryHtml(story,members){
  const links=story.members.map(name=>{
    const xs=bondMemberMatches(members,name).filter(m=>!m.deletedAt);
    return xs.length===1?'<button type="button" class="ghost-btn tiny" data-bond-person="'+esc(xs[0].id)+'">'+esc(name)+'</button>':'<span class="badge">'+esc(name)+'</span>';
  }).join('');
  return '<article class="card bond-story"><h3>'+esc(story.title)+'</h3><div class="badges"><span class="badge '+(story.status==='確認済み'?'ok':'warn')+'">'+esc(story.status)+'</span></div><p>'+bondLines(story.body)+'</p><div class="tiny muted">関わる人物</div><div class="actions">'+links+'</div>'+(story.unresolved?'<p class="bond-note">'+bondLines(story.unresolved)+'</p>':'')+'</article>';
}
function bondGroupsHtml(members,personId,query){
  const person=members.find(m=>m.id===personId);
  const groups=MOA_BOND_EDITION.groups.filter(g=>(!person||g.members.some(n=>bondMemberMatches([person],n).length))&&bondQueryMatches(JSON.stringify(g),query));
  if(!groups.length)return '';
  return '<details class="card bond-groups"><summary>所属・仲間のまとまり（'+groups.length+'）</summary>'+groups.map(g=>'<section><h3>'+esc(g.title)+'</h3><p>'+esc(g.description)+'</p><div class="actions">'+g.members.map(n=>{const x=bondMemberMatches(members,n).filter(m=>!m.deletedAt);return x.length===1?'<button type="button" class="ghost-btn tiny" data-bond-person="'+esc(x[0].id)+'">'+esc(n)+'</button>':'<span class="badge">'+esc(n)+'</span>';}).join('')+'</div></section>').join('')+'</details>';
}
