
// ---- Android/PWA back-button history support ----
const NAV_STATE_KEY='moaNavState';
function navSnapshot(){return {moaNav:true,view:state.view,memberId:state.memberId,memberTab:state.memberTab,search:state.search,job:state.job,month:state.month instanceof Date?state.month.toISOString():state.month,selectedDate:state.selectedDate};}
function applyNavSnapshot(s){if(!s||!s.moaNav)return;state.view=s.view||'home';state.memberId=s.memberId||null;state.memberTab=s.memberTab||'profile';state.search=s.search||'';state.job=s.job||'すべて';state.month=s.month?new Date(s.month):new Date();state.selectedDate=s.selectedDate||toDateKey(new Date());document.querySelectorAll('.nav-btn').forEach(b=>b.classList.toggle('active',b.dataset.view===state.view));}
function sameNav(a,b){return !!(a&&b&&a.moaNav&&b.moaNav&&a.view===b.view&&a.memberId===b.memberId&&a.memberTab===b.memberTab&&a.search===b.search&&a.job===b.job&&a.selectedDate===b.selectedDate);}
function pushCurrentNav(){const snap=navSnapshot();if(!sameNav(history.state,snap))history.pushState(snap,'',location.href);}

const _historyRenderMemberDetail=renderMemberDetail;
renderMemberDetail=async function(id){await _historyRenderMemberDetail(id);const back=document.getElementById('backMembers');if(back)back.onclick=()=>{if(history.length>1)history.back();else{state.memberId=null;state.view='members';render();}};};

document.addEventListener('click',e=>{
  const t=e.target.closest?.('[data-view],[data-member],[data-mtab],[data-prod-member]');
  if(!t)return;
  setTimeout(()=>pushCurrentNav(),0);
});

window.addEventListener('popstate',e=>{
  if(document.querySelector('.modal-backdrop'))closeModal();
  if(e.state?.moaNav){applyNavSnapshot(e.state);render();}
});

// Replace the current launcher entry with the current in-app screen.
setTimeout(()=>{if(!history.state?.moaNav)history.replaceState(navSnapshot(),'',location.href);},0);

// ---- Member confirmation icon support (kept separate from angle masters) ----
const PROFILE_CATEGORY='プロフィール写真';

function profileImageFor(images=[]){
  return images
    .filter(i=>!i.deletedAt&&i.category===PROFILE_CATEGORY&&i.dataUrl)
    .sort((a,b)=>String(b.updatedAt||b.createdAt||'').localeCompare(String(a.updatedAt||a.createdAt||'')))[0]||null;
}

function memberAvatarHtml(m,image,size=''){
  if(image?.dataUrl)return `<img class="avatar ${size}" src="${esc(image.dataUrl)}" alt="${esc(m.name)}の確認用アイコン写真" loading="lazy">`;
  return `<div class="avatar ${size} avatar-placeholder">${esc(initials(m.name))}</div>`;
}

memberCard=function(m,profileImage=null){
  return `<div class="member-card" data-member="${m.id}">${memberAvatarHtml(m,profileImage)}<div class="grow"><div class="member-name">${esc(m.name)} ${m.favorite?'★':''}</div><div class="muted tiny">${m.age?`${m.age}歳 ・ `:''}${esc(m.job||'職業未設定')}</div><div class="badges"><span class="badge ${m.status==='確定'?'ok':'warn'}">${esc(m.status||'候補')}</span>${m.tags?`<span class="badge">${esc(m.tags)}</span>`:''}</div></div><span class="gold">›</span></div>`;
};

renderMembers=async function(){
  const members=await activeMembers();
  const images=await activeImages();
  const jobs=['すべて',...new Set(members.map(m=>m.job).filter(Boolean))];
  const filtered=members.filter(m=>{const q=state.search.trim().toLowerCase();const hit=!q||[m.name,m.kana,m.job,m.affiliation].some(v=>String(v||'').toLowerCase().includes(q));const job=state.job==='すべて'||m.job===state.job;return hit&&job;});
  app.innerHTML=`
    <input id="memberSearch" class="search" placeholder="名前・職業・所属で検索" value="${esc(state.search)}" />
    <div class="chips">${jobs.map(j=>`<button class="chip ${state.job===j?'active':''}" data-job="${esc(j)}">${esc(j)}</button>`).join('')}</div>
    <div class="section-title"><h2>メンバー</h2><small>${filtered.length}人</small></div>
    <div>${filtered.length?filtered.map(m=>memberCard(m,profileImageFor(images.filter(i=>i.memberId===m.id)))).join(''):`<div class="empty">該当するメンバーがいません</div>`}</div>
  `;
  document.getElementById('memberSearch').addEventListener('input',e=>{state.search=e.target.value;renderMembers()});
  document.querySelectorAll('[data-job]').forEach(b=>b.addEventListener('click',()=>{state.job=b.dataset.job;renderMembers()}));
  document.querySelectorAll('[data-member]').forEach(b=>b.addEventListener('click',()=>{state.memberId=b.dataset.member;state.memberTab='profile';render()}));
};

async function iconPhotoModal(memberId){
  const m=await getOne('members',memberId);
  if(!m){toast('人物が見つかりません');return;}
  const images=(await activeImages()).filter(i=>i.memberId===memberId);
  const current=profileImageFor(images);
  const faceFront=images.find(i=>i.category==='顔8方向'&&i.angle==='正面0°');
  showModal('確認用アイコン写真',`<form id="iconPhotoForm" class="form">
    <div class="icon-photo-note">人物一覧と、この画面の左上に表示する確認用写真です。顔8方向や角度画像には登録されません。</div>
    ${current?`<img class="icon-photo-preview" src="${esc(current.dataUrl)}" alt="${esc(m.name)}の現在のアイコン写真">`:''}
    <div class="field"><label>${current?'別の写真に変更':'写真を選ぶ'} *</label><input name="file" type="file" accept="image/*" required></div>
    <button class="gold-btn" type="submit">${current?'アイコン写真を変更':'アイコン写真を設定'}</button>
    ${!current&&faceFront?`<button class="ghost-btn" id="useFrontAsIcon" type="button">誤って正面0°に入れた写真をアイコンへ移す</button>`:''}
  </form>`);
  document.getElementById('iconPhotoForm').onsubmit=async e=>{
    e.preventDefault();
    const file=new FormData(e.currentTarget).get('file');
    const dataUrl=await fileToDataURL(file);
    if(current){
      current.dataUrl=dataUrl;
      current.updatedAt=now();
      await put('images',current);
      await log('アイコン写真変更','image',current.id,m.name);
    }else{
      const obj={id:uid('img'),memberId,category:PROFILE_CATEGORY,angle:'',status:'採用・安定',basis:'使用しない',memo:'人物確認用アイコン',dataUrl,createdAt:now(),updatedAt:now(),deletedAt:null};
      await put('images',obj);
      await log('アイコン写真設定','image',obj.id,m.name);
    }
    closeModal();
    toast('確認用アイコン写真を保存しました');
    renderMemberDetail(memberId);
  };
  const useFront=document.getElementById('useFrontAsIcon');
  if(useFront)useFront.onclick=()=>moveImageToProfile(faceFront.id);
}

async function moveImageToProfile(imageId){
  const image=await getOne('images',imageId);
  if(!image||image.deletedAt){toast('画像が見つかりません');return;}
  if(!confirm('誤って「顔正面0°」に入ったこの写真を、確認用アイコンへ移しますか？\n顔正面0°の枠からは外れます。'))return;
  image.category=PROFILE_CATEGORY;
  image.angle='';
  image.status='採用・安定';
  image.basis='使用しない';
  image.memo=image.memo||'人物確認用アイコン';
  image.updatedAt=now();
  await put('images',image);
  await log('画像の登録先変更','image',image.id,'顔正面0° → 確認用アイコン');
  closeModal();
  toast('確認用アイコンへ移しました');
  renderMemberDetail(image.memberId);
}

const _profileRenderMemberDetail=renderMemberDetail;
renderMemberDetail=async function iconRenderMemberDetail(id){
  await _profileRenderMemberDetail(id);
  const m=await getOne('members',id);
  if(!m)return;
  const images=(await activeImages()).filter(i=>i.memberId===id);
  const profile=profileImageFor(images);
  let avatar=document.querySelector('.detail-head .avatar');
  if(!avatar)return;
  if(profile){
    avatar.outerHTML=memberAvatarHtml(m,profile,'large');
    avatar=document.querySelector('.detail-head .avatar');
  }
  const picker=document.createElement('button');
  picker.type='button';
  picker.className='avatar-picker';
  picker.setAttribute('aria-label',`${m.name}の確認用アイコン写真を${profile?'変更':'設定'}`);
  picker.title='タップして確認用写真を設定';
  avatar.replaceWith(picker);
  picker.appendChild(avatar);
  const badge=document.createElement('span');
  badge.className='avatar-picker-badge';
  badge.textContent=profile?'変更':'写真';
  picker.appendChild(badge);
  picker.onclick=()=>iconPhotoModal(id);
};

imageModal=async function(memberId,preferredCategory=''){
  const members=await activeMembers();
  const defaultCategory=preferredCategory||'顔8方向';
  const categories=['顔8方向','体型8方向',...ALBUMS];
  showModal('画像を追加',`<form id="imageForm" class="form"><div class="field"><label>人物</label><select name="memberId" required>${members.map(m=>`<option value="${m.id}" ${m.id===memberId?'selected':''}>${esc(m.name)}</option>`).join('')}</select></div><div class="field"><label>画像 *</label><input name="file" type="file" accept="image/*" required></div><div class="field"><label>登録先</label><select name="category" id="imageCategory">${categories.map(x=>`<option ${x===defaultCategory?'selected':''}>${x}</option>`).join('')}</select><div id="imageCategoryHint" class="tiny muted" style="margin-top:7px"></div></div><div class="field" id="angleField"><label>角度</label><select name="angle">${FACE_ANGLES.map(x=>`<option>${x}</option>`).join('')}</select></div><div class="field-row"><div class="field"><label>状態</label><select name="status">${IMAGE_STATUS.map(x=>`<option>${x}</option>`).join('')}</select></div><div class="field" id="basisField"><label>生成基準の強さ</label><select name="basis">${BASIS.map(x=>`<option>${x}</option>`).join('')}</select></div></div><div class="field"><label>評価メモ</label><textarea name="memo" placeholder="本人固定・角度・髪・体型など"></textarea></div><button class="gold-btn" type="submit">画像を保存</button></form>`);
  const cat=document.getElementById('imageCategory');
  const angle=document.getElementById('angleField');
  const basis=document.getElementById('basisField');
  const hint=document.getElementById('imageCategoryHint');
  const syncCategory=()=>{
    const isMaster=['顔8方向','体型8方向'].includes(cat.value);
    angle.style.display=isMaster?'grid':'none';
    basis.style.display='grid';
    hint.textContent=isMaster?'角度マスターの登録先です。':'アルバムへ追加します。';
  };
  cat.onchange=syncCategory;
  syncCategory();
  document.getElementById('imageForm').onsubmit=async e=>{
    e.preventDefault();
    const f=new FormData(e.currentTarget);
    const file=f.get('file');
    const dataUrl=await fileToDataURL(file);
    const category=f.get('category');
    const isProfile=category===PROFILE_CATEGORY;
    const obj={id:uid('img'),memberId:f.get('memberId'),category,angle:['顔8方向','体型8方向'].includes(category)?f.get('angle'):'',status:isProfile?'採用・安定':f.get('status'),basis:isProfile?'使用しない':f.get('basis'),memo:f.get('memo'),dataUrl,createdAt:now(),updatedAt:now(),deletedAt:null};
    await put('images',obj);
    await log('画像追加','image',obj.id,`${category} ${obj.angle||''}`);
    closeModal();
    toast(isProfile?'プロフィール写真を保存しました':'画像を保存しました');
    if(isProfile)state.memberTab='profile';
    render();
  };
};
