const PROFILE_CATEGORY='プロフィール写真';

function profileImageFor(images=[]){
  return images
    .filter(i=>!i.deletedAt&&i.category===PROFILE_CATEGORY&&i.dataUrl)
    .sort((a,b)=>String(b.updatedAt||b.createdAt||'').localeCompare(String(a.updatedAt||a.createdAt||'')))[0]||null;
}

function memberAvatarHtml(m,image,size=''){
  if(image?.dataUrl)return `<img class="avatar ${size}" src="${esc(image.dataUrl)}" alt="${esc(m.name)}の確認用アイコン写真" loading="lazy">`;
  return `<div class="avatar ${size} avatar-placeholder">${esc(initials(m.name))}</div>`;
}

memberCard=function(m,profileImage=null){
  return `<div class="member-card" data-member="${m.id}">${memberAvatarHtml(m,profileImage)}<div class="grow"><div class="member-name">${esc(m.name)} ${m.favorite?'★':''}</div><div class="muted tiny">${m.age?`${m.age}歳 ・ `:''}${esc(m.job||'職業未設定')}</div><div class="badges"><span class="badge ${m.status==='確定'?'ok':'warn'}">${esc(m.status||'候補')}</span>${m.tags?`<span class="badge">${esc(m.tags)}</span>`:''}</div></div><span class="gold">›</span></div>`;
};

async function moveImageToProfile(imageId){
  const image=await getOne('images',imageId);
  if(!image||image.deletedAt){toast('画像が見つかりません');return;}
  if(!confirm('誤って「顔正面0°」に入ったこの写真を、確認用アイコンへ移しますか？\n顔正面0°の枠からは外れます。'))return;
  image.category=PROFILE_CATEGORY;
  image.angle='';
  image.status='採用・安定';
  image.basis='使用しない';
  image.memo=image.memo||'人物確認用アイコン';
  image.updatedAt=now();
  await put('images',image);
  await log('画像の登録先変更','image',image.id,'顔正面0° → 確認用アイコン');
  closeModal();
  toast('確認用アイコンへ移しました');
  renderMemberDetail(image.memberId);
}

// ---- Angle master automatic orientation check (MediaPipe Pose Landmarker) ----
// Assistive only: mismatches warn, but the user can still save.
const ANGLE_CHECK_ENGINE='mediapipe-pose-1.0.1-v1';
const ANGLE_CHECK_CDN='https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@1.0.1/vision_bundle.mjs';
const ANGLE_CHECK_WASM='https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@1.0.1/wasm';
const ANGLE_CHECK_MODEL='https://storage.googleapis.com/mediapipe-models/pose_landmarker/pose_landmarker_lite/float16/1/pose_landmarker_lite.task';
let anglePoseLandmarkerPromise=null;

function angleLandmarkScore(p){
  if(!p)return 0;
  if(Number.isFinite(p.visibility))return p.visibility;
  if(Number.isFinite(p.presence))return p.presence;
  return 1;
}
function angleAverage(values){const xs=values.filter(Number.isFinite);return xs.length?xs.reduce((a,b)=>a+b,0)/xs.length:0;}
function angleImageFromDataUrl(dataUrl){
  return new Promise((resolve,reject)=>{
    const img=new Image();
    img.onload=()=>resolve(img);
    img.onerror=()=>reject(new Error('画像を読み込めませんでした'));
    img.src=dataUrl;
  });
}
async function getAnglePoseLandmarker(){
  if(!anglePoseLandmarkerPromise){
    anglePoseLandmarkerPromise=(async()=>{
      const visionModule=await import(ANGLE_CHECK_CDN);
      const vision=await visionModule.FilesetResolver.forVisionTasks(ANGLE_CHECK_WASM);
      return visionModule.PoseLandmarker.createFromOptions(vision,{
        baseOptions:{modelAssetPath:ANGLE_CHECK_MODEL},
        runningMode:'IMAGE',
        numPoses:1,
        minPoseDetectionConfidence:0.45,
        minPosePresenceConfidence:0.45,
        minTrackingConfidence:0.45
      });
    })();
  }
  try{return await anglePoseLandmarkerPromise;}
  catch(err){anglePoseLandmarkerPromise=null;throw err;}
}

function classifyFaceMasterAngle(lm,targetAngle){
  const nose=lm?.[0],leftEye=lm?.[2],rightEye=lm?.[5],leftEar=lm?.[7],rightEar=lm?.[8];
  if(!nose||!leftEye||!rightEye)return null;

  const leftEyeScore=angleLandmarkScore(leftEye),rightEyeScore=angleLandmarkScore(rightEye);
  const leftEarScore=angleLandmarkScore(leftEar),rightEarScore=angleLandmarkScore(rightEar);
  const eyeMid=(Number(leftEye.x)+Number(rightEye.x))/2;
  const eyeSpan=Math.abs(Number(leftEye.x)-Number(rightEye.x));
  const delta=Number(nose.x)-eyeMid;
  if(!Number.isFinite(delta)||!Number.isFinite(eyeSpan)||eyeSpan<0.003)return null;

  const ratio=Math.abs(delta)/Math.max(eyeSpan,0.006);
  const direction=delta<0?'左':'右';
  const sideOcclusion=Math.max(Math.abs(leftEyeScore-rightEyeScore),Math.abs(leftEarScore-rightEarScore));
  const predicted=ratio<0.20?'正面0°':ratio<0.55?direction+'前45°':direction+'90°';
  const detail='顔向き比 '+ratio.toFixed(2)+'・左右の見え方差 '+sideOcclusion.toFixed(2)+'（顔8方向は肩・胴体を判定から除外）';

  if(targetAngle==='正面0°'){
    if(ratio<0.20&&sideOcclusion<0.35)return {status:'match',predicted:'正面0°',confidence:'高',reason:detail};
    if(ratio<0.30)return {status:'uncertain',predicted,confidence:'中',reason:detail};
    return {status:'mismatch',predicted,confidence:'中',reason:detail};
  }

  if(targetAngle==='左前45°'||targetAngle==='右前45°'){
    const wanted=targetAngle.startsWith('左')?'左':'右';
    if(direction!==wanted&&ratio>=0.20)return {status:'mismatch',predicted,confidence:'中',reason:detail};
    if(ratio>=0.20&&ratio<0.58&&sideOcclusion<0.60)return {status:'match',predicted:targetAngle,confidence:ratio>=0.28?'高':'中',reason:detail};
    if(ratio>=0.15&&ratio<0.75)return {status:'uncertain',predicted,confidence:'中',reason:detail};
    return {status:'mismatch',predicted,confidence:'中',reason:detail};
  }

  if(targetAngle==='左90°'||targetAngle==='右90°'){
    const wanted=targetAngle.startsWith('左')?'左':'右';
    if(direction!==wanted&&ratio>=0.20)return {status:'mismatch',predicted,confidence:'中',reason:detail};
    // A profile face may still have a front-facing torso. For face masters,
    // strong nose offset OR one-sided eye/ear visibility is enough for 90°.
    if(ratio>=0.50||sideOcclusion>=0.38)return {status:'match',predicted:targetAngle,confidence:(ratio>=0.65||sideOcclusion>=0.52)?'高':'中',reason:detail};
    if(ratio>=0.30)return {status:'uncertain',predicted,confidence:'中',reason:detail};
    return {status:'mismatch',predicted,confidence:'中',reason:detail};
  }
  return null;
}

function classifyAnglePose(result,targetAngle,category=''){
  const lm=result?.landmarks?.[0];
  const world=result?.worldLandmarks?.[0];
  if(!lm||lm.length<13)return {status:'uncertain',predicted:'判定不能',confidence:'低',reason:'人物の姿勢ランドマークを十分に検出できませんでした'};

  if(category==='顔8方向'&&['正面0°','左前45°','右前45°','左90°','右90°'].includes(targetAngle)){
    const faceResult=classifyFaceMasterAngle(lm,targetAngle);
    if(faceResult)return faceResult;
  }

  const nose=lm[0],leftEye=lm[2],rightEye=lm[5],leftEar=lm[7],rightEar=lm[8],leftShoulder=lm[11],rightShoulder=lm[12];
  const shoulderVis=Math.min(angleLandmarkScore(leftShoulder),angleLandmarkScore(rightShoulder));
  if(shoulderVis<0.25)return {status:'uncertain',predicted:'判定不能',confidence:'低',reason:'両肩を十分に検出できませんでした'};

  const faceVis=angleAverage([nose,leftEye,rightEye,leftEar,rightEar].map(angleLandmarkScore));
  let yawAbs=null;
  let direction='';

  if(world?.[11]&&world?.[12]){
    const dx=Math.abs(Number(world[11].x)-Number(world[12].x));
    const dz=Number(world[11].z)-Number(world[12].z);
    if(Number.isFinite(dx)&&Number.isFinite(dz)){
      yawAbs=Math.atan2(Math.abs(dz),Math.max(dx,0.001))*180/Math.PI;
      direction=dz<0?'左':'右';
    }
  }

  if(faceVis>=0.32&&nose&&leftEye&&rightEye&&angleLandmarkScore(leftEye)>=0.2&&angleLandmarkScore(rightEye)>=0.2){
    const eyeMid=(Number(leftEye.x)+Number(rightEye.x))/2;
    const delta=Number(nose.x)-eyeMid;
    if(Number.isFinite(delta)&&Math.abs(delta)>0.008)direction=delta<0?'左':'右';
  }

  if(!Number.isFinite(yawAbs)){
    const shoulderSpan=Math.abs(Number(leftShoulder.x)-Number(rightShoulder.x));
    if(nose&&Number.isFinite(nose.x)&&shoulderSpan>0.01){
      const mid=(Number(leftShoulder.x)+Number(rightShoulder.x))/2;
      const offset=Math.abs(Number(nose.x)-mid)/shoulderSpan;
      yawAbs=Math.min(90,offset*160);
      if(!direction&&offset>0.05)direction=Number(nose.x)<mid?'左':'右';
    }
  }
  if(!Number.isFinite(yawAbs))return {status:'uncertain',predicted:'判定不能',confidence:'低',reason:'回転量を計算できませんでした'};

  let predicted='';
  if(yawAbs<24){
    predicted=faceVis>=0.30?'正面0°':'真後ろ180°';
  }else if(yawAbs<68){
    if(!direction)return {status:'uncertain',predicted:'45°付近',confidence:'低',reason:'左右方向を安定して判定できませんでした'};
    predicted=direction+(faceVis>=0.30?'前45°':'後ろ45°');
  }else{
    if(!direction)return {status:'uncertain',predicted:'90°付近',confidence:'低',reason:'左右方向を安定して判定できませんでした'};
    predicted=direction+'90°';
  }

  const nearBoundary=Math.abs(yawAbs-24)<7||Math.abs(yawAbs-68)<9;
  const confidence=(nearBoundary||shoulderVis<0.55||(predicted.indexOf('後ろ')>=0&&faceVis>0.45)||(predicted==='真後ろ180°'&&faceVis>0.22))?'中':'高';
  const status=nearBoundary?'uncertain':(predicted===targetAngle?'match':'mismatch');
  const reason=nearBoundary?'回転量が角度区分の境界付近です（推定 '+Math.round(yawAbs)+'°）':'推定 '+Math.round(yawAbs)+'°・顔の見えやすさ '+Math.round(faceVis*100)+'%';
  return {status,predicted,confidence,reason};
}
async function inspectAngleImage(dataUrl,targetAngle,category=''){
  try{
    const pair=await Promise.all([getAnglePoseLandmarker(),angleImageFromDataUrl(dataUrl)]);
    const result=pair[0].detect(pair[1]);
    return {...classifyAnglePose(result,targetAngle,category),target:targetAngle,category,engine:ANGLE_CHECK_ENGINE,checkedAt:now()};
  }catch(err){
    console.warn('angle check unavailable',err);
    return {status:'unavailable',predicted:'判定できず',confidence:'低',reason:'自動角度チェックを起動できませんでした',target:targetAngle,category,engine:ANGLE_CHECK_ENGINE,checkedAt:now()};
  }
}
async function confirmAngleCheckBeforeSave(dataUrl,targetAngle,category=''){
  toast('角度を自動チェック中…');
  const result=await inspectAngleImage(dataUrl,targetAngle,category);
  if(result.status==='match'){
    toast('角度チェック：'+targetAngle+' と一致');
    return {proceed:true,result};
  }
  if(result.status==='mismatch'){
    const proceed=confirm('⚠ 角度チェック\n\n登録先：'+targetAngle+'\n自動判定：'+result.predicted+'\n信頼度：'+result.confidence+'\n'+result.reason+'\n\n角度が違う可能性があります。\nそれでもこの枠へ登録しますか？');
    return {proceed,result};
  }
  const proceed=confirm('⚠ 角度チェックは確定できませんでした\n\n登録先：'+targetAngle+'\n判定：'+result.predicted+'\n'+result.reason+'\n\n目視で角度を確認してください。\nそれでも登録しますか？');
  return {proceed,result};
}

    </div>
    <div class="field"><label>評価メモ</label><textarea name="memo" placeholder="本人固定・角度・髪・体型など">${esc(current?.memo||'')}</textarea></div>
    <button class="gold-btn" type="submit">${current?'この角度の画像を差し替える':'この角度へ画像を追加'}</button>
  </form>`);
  document.getElementById('angleSlotForm').onsubmit=async e=>{
    e.preventDefault();
    if(current&&!confirm(`${angle}の現在画像を、新しく選んだ画像へ差し替えますか？`))return;
    const f=new FormData(e.currentTarget);
    const dataUrl=await fileToDataURL(f.get('file'));
    const checked=await confirmAngleCheckBeforeSave(dataUrl,angle,category);
    if(!checked.proceed)return;
    if(current){
      current.dataUrl=dataUrl;
      current.status=f.get('status');
      current.basis=f.get('basis');
      current.memo=f.get('memo');
      current.angleCheck=checked.result;
      current.updatedAt=now();
      await put('images',current);
      await log('角度画像差し替え','image',current.id,`${category} ${angle}`);
    }else{
      const obj={id:uid('img'),memberId,category,angle,status:f.get('status'),basis:f.get('basis'),memo:f.get('memo'),dataUrl,angleCheck:checked.result,createdAt:now(),updatedAt:now(),deletedAt:null};
      await put('images',obj);
      await log('角度画像追加','image',obj.id,`${category} ${angle}`);
    }
    closeModal();
    toast(`${angle}の画像を保存しました`);
    renderMemberDetail(memberId);
  };
}

};
