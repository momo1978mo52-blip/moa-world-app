async function exportData(){
  try{
    const data={version:'9.1',exportedAt:now(),stores:{}};
    for(const s of STORES)data.stores[s]=await getAll(s);
    const json=JSON.stringify(data,null,2);
    const filename=`moa-world-backup-${toDateKey(new Date())}.json`;
    const file=new File([json],filename,{type:'application/json'});
    if(navigator.share&&navigator.canShare&&navigator.canShare({files:[file]})){
      try{
        await navigator.share({files:[file],title:'MOA WORLD バックアップ'});
        toast('バックアップを保存・共有しました');
        return;
      }catch(err){
        if(err?.name==='AbortError'){toast('バックアップ保存をキャンセルしました');return;}
        console.warn('Web Share failed, falling back to download',err);
      }
    }
    const blob=new Blob([json],{type:'application/json'});
    const url=URL.createObjectURL(blob);
    const a=document.createElement('a');
    a.href=url;
    a.download=filename;
    a.style.display='none';
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(()=>URL.revokeObjectURL(url),5000);
    toast('バックアップの保存を開始しました');
  }catch(err){
    console.error(err);
    alert('バックアップの書き出しに失敗しました: '+(err?.message||err));
  }
}
