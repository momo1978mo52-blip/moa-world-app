async function exportData(includeImages=false){
  showModal(includeImages?'画像込み完全バックアップ':'軽量バックアップ',
    '<div id="backupBuildStatus" class="empty">バックアップを作成しています…</div>');
  const status=document.getElementById('backupBuildStatus');
  try{
    const data={version:'9.2',backupKind:includeImages?'full':'light',includesImageData:!!includeImages,exportedAt:now(),stores:{}};
    for(const s of STORES){
      const rows=await getAll(s);
      if(s==='images'&&!includeImages){
        data.stores[s]=rows.map(row=>{
          const copy={...row};
          if('dataUrl' in copy)delete copy.dataUrl;
          return copy;
        });
      }else{
        data.stores[s]=rows;
      }
    }
    const json=JSON.stringify(data);
    const bytes=new Blob([json]).size;
    const sizeText=bytes<1024*1024?`${Math.max(1,Math.round(bytes/1024))} KB`:`${(bytes/1024/1024).toFixed(1)} MB`;
    const filename=`moa-world-${includeImages?'full':'light'}-backup-${toDateKey(new Date())}.json`;
    status.className='';
    status.innerHTML=`<div class="card"><b>バックアップ準備完了</b><div class="tiny muted" style="margin-top:8px">${includeImages?'画像本体を含む完全版':'画像本体を除いた軽量版'} ・ ${sizeText}</div></div><button class="gold-btn" id="saveBackupFile" type="button">端末に保存</button><button class="ghost-btn" id="cancelBackupFile" type="button">キャンセル</button>`;
    document.getElementById('saveBackupFile').onclick=()=>{
      try{
        const blob=new Blob([json],{type:'application/json'});
        const url=URL.createObjectURL(blob);
        const a=document.createElement('a');
        a.href=url;
        a.download=filename;
        a.rel='noopener';
        a.style.display='none';
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(()=>URL.revokeObjectURL(url),30000);
        closeModal();
        toast('バックアップを保存しました');
      }catch(err){
        console.error(err);
        alert('保存に失敗しました: '+(err?.message||err));
      }
    };
    document.getElementById('cancelBackupFile').onclick=closeModal;
  }catch(err){
    console.error(err);
    if(status){
      status.className='empty';
      status.innerHTML=`バックアップ作成に失敗しました。<br><small>${esc(err?.message||err)}</small>`;
    }else{
      alert('バックアップ作成に失敗しました: '+(err?.message||err));
    }
  }
}

function readBackupFileText(file){
  if(file&&typeof file.text==='function')return file.text();
  return new Promise((resolve,reject)=>{
    const r=new FileReader();
    r.onload=()=>resolve(String(r.result||''));
    r.onerror=()=>reject(r.error||new Error('ファイルを読み込めませんでした'));
    r.readAsText(file);
  });
}
